# PICT pairwise model for resolveProductBookAssets (web/src/lib/book-assets.ts).
# Factors:
#   house: creator is the house (creator_is_house true). Only the house may use slug-keyed registry overrides
#          (GLOBAL.storefront.book_assets). A non-house product takes books/*.json assets only by exact product_id identity.
#   override: GLOBAL.storefront.book_assets entry keyed by slug (slug) or no entry (none). The resolver does not
#          key overrides by product id: that rekey was declined by the founder (2026-11-21), so no id value here.
#   book_file: the product row's book_registry: the sample registry file (sample_book, is_sample true), the non-sample
#          registry file (duane_book, is_sample false), or none. With none and a slug override the override's file applies.
#   identity: product.id against the resolved file's product_id: match (equal), mismatch (another id), missing (no id).
#   sample_source: where an explicit is_sample flag comes from: the override entry (override_true / override_false),
#          the registry file (registry_true), or neither (absent).
# Constraints:
#   no file (book_file none) gives no identity to check, so identity is missing.
#   an override flag needs a slug override entry, so override_true / override_false need override = slug.
#   a registry flag needs the sample registry file.
# Run: /usr/local/bin/pict docs/testing/pairwise/book_assets_resolver.model.txt > docs/testing/pairwise/book_assets_resolver.rows.tsv

house: yes, no
override: slug, none
book_file: sample_book, duane_book, none
identity: match, mismatch, missing
sample_source: override_true, override_false, registry_true, absent

IF [book_file] = "none" THEN [identity] = "missing";
IF [sample_source] IN {"override_true", "override_false"} THEN [override] = "slug";
IF [sample_source] = "registry_true" THEN [book_file] = "sample_book";
