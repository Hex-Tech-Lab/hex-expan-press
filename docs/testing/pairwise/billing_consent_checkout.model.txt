# PICT pairwise model for billing / consent / checkout parameters of /api/billing/checkout and webhook consent persistence.
# Rev 3 (paddle-option-a): adds provider_path (overlay vs redirect) and asserts buyer_consent true/false reaching the webhook.
# Rev 4 (house-title-consent-exemption, 2026-11-19): adds creator_kind (creator vs house). A house title skips the creator consents
#   C1-C3 (decision and consent_kind have no effect on it) and must still match its pinned release hash (house_release_pins).
# Run: /usr/local/bin/pict docs/testing/pairwise/billing_consent_checkout.model.txt > docs/testing/pairwise/billing_consent_checkout.rows.tsv
#
# Parameters:
#   consent_kind  : consent kind under evaluation (the other two gate kinds remain "given" in row setup)
#   decision      : head state of consent_kind ("forked" = given + refused heads; "resolved_by_set" = fork retired via supersedes_ids)
#   provider      : product_rails.provider. paddle and creem are routable in the test world (creem is switched off in production
#                   settings_registry and is enabled here by a test-only override); polar, fungies, lemonsqueezy are switched off
#   provider_path : checkout delivery path (overlay = POST Accept: application/json opening Paddle.js overlay; redirect = GET 302)
#   checkout_mode : products.checkout_mode
#   sandbox_flag  : GLOBAL.payments.sandbox_checkout_enabled (default false; true is an opt-in)
#   vercel_env    : process.env.VERCEL_ENV (unset_node_prod / unset_node_test: VERCEL_ENV unset, NODE_ENV production / test)
#   product_state : product lookup, C2 release hash, and rail presence (active_rail = one active rail row)
#   rail_mode     : host class of the active rail's checkout_url (live host, or sandbox host)
#   buyer_email   : buyer_email parameter (absent; valid; invalid shape; typo_suggested = valid shape with a likely domain typo)
#   buyer_consent : buyer_consent parameter (checked = "true", unchecked = "false"; carried on overlay/creem and persisted by webhook)
#   buyer_country : x-vercel-ip-country header (us = US; us_territory = PR, VI, GU, AS, MP; non_us = outside the allow list)
#   creator_kind  : owner of the product (creator = external creator, C1-C3 gate applies; house = creators.is_house, consent gate skipped)

consent_kind: C1_data_accuracy, C2_release_approval, C3_revenue_split
decision: given, refused, forked, resolved_by_set
provider: paddle, polar, creem, fungies, lemonsqueezy
provider_path: overlay, redirect
checkout_mode: live, sandbox, gated
sandbox_flag: false, true
vercel_env: production, preview, development, unset_node_prod, unset_node_test
product_state: active_rail, no_rails, release_hash_mismatch, bare_slug_ambiguous, bare_slug_single
rail_mode: live, sandbox
buyer_email: absent, valid, invalid, typo_suggested
buyer_consent: checked, unchecked
buyer_country: us, us_territory, non_us
creator_kind: creator, house

# Only Paddle supports the overlay path; other providers keep their redirect path.
IF [provider] <> "paddle" THEN [provider_path] = "redirect";
# States that never reach a rail: the rail's provider, path and host class cannot change the outcome, so they are pinned.
IF [product_state] IN {"no_rails", "release_hash_mismatch", "bare_slug_ambiguous", "bare_slug_single"} THEN [rail_mode] = "live" AND [provider] = "paddle" AND [provider_path] = "redirect";
