# Length values are symbolic bands; the test maps them to HANDLE_MIN / mid / HANDLE_MAX / HANDLE_MAX + 1.

last_char: a, z, 0, 9
separator_count: 0, 1, 2
separator_position: start, middle, none

case: lower, camel

IF [separator_count] = 2 AND [separator_position] = "middle" THEN [length_band] <> "min";
